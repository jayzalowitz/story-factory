% S01
\scene{INT. MUNICIPAL RECORDS DEPOT -- LEDGER ROOM -- TUESDAY MORNING}
\action{CLARA VENN, 54, puts a saucer under a cabinet leg. The cabinet stops rocking. She removes a receipt from the saucer, reads it, files it.}
\action{Outside the high windows: gulls. Inside: twenty-seven years of municipal paper, sorted for a journey nobody wants to make. A sign says PUBLIC READING ROOM. Someone has crossed out PUBLIC.}
\action{TOM KEST, 60, wheels in a trolley with one square wheel.}
\character{TOM}
\dialogue{They've taken the kettle.}
\character{CLARA}
\dialogue{It's an electrical appliance. It went yesterday.}
\character{TOM}
\dialogue{It had water in it.}
\character{CLARA}
\dialogue{Then they've taken the reservoir too.}
\action{He presents a crate. Clara checks the six-digit seal against a book, not against the label. He waits with the practiced patience of a man who will wait no longer after Thursday.}
\character{TOM}
\dialogue{My last box.}
\character{CLARA}
\dialogue{Your last ordinary box. We still have exceptions.}
\character{TOM}
\dialogue{I'm not an exception from nine on Thursday.}
\action{He puts his pension acceptance form beside the crate. Her signature already sits on it. She shifts the form out of the way before checking the next number.}
\action{A delivery horn makes two polite notes, then one thoroughly impolite one. Tom goes. Clara circles a transposed digit in red. Below her hand: TRANSFER COMPLETE, with an empty space for her signature.}
% S02
\scene{EXT. MUNICIPAL RECORDS DEPOT -- YARD -- MORNING}

\dialogue{Your manifest says vehicle access.}
\character{CLARA}
\dialogue{You appear to have accessed it.}

\character{JORY}
\dialogue{You ought to put that on the manifest.}
\character{CLARA}
\dialogue{Then they'd send a larger van.}
\action{Jory hands her a terminal. The screen offers ACCEPT ALL. Clara scrolls down to INDIVIDUAL ITEMS.}

\dialogue{Forty-eight hours. Nine Thursday, keys to the contractor. Nine oh one, I stop existing.}
\character{CLARA}
\dialogue{Temporary contract?}
\character{JORY}
\dialogue{Permanent contract conditional on satisfactory closure. They've found a way to make permanence temporary.}
\action{Clara signs only for the terminal. Jory watches the small distinction.}
\character{JORY}
\dialogue{You're the completion signatory?}
\character{CLARA}
\dialogue{I'm the person who can tell you what's here.}
\character{JORY}
\dialogue{That's the one I need.}
\action{She produces two pairs of gloves. Clara takes one.}
% S03

\dialogue{Twenty-nine. Two were condemned.}
\character{CLARA}
\dialogue{By whom?}

\dialogue{By the paper conservator whose fee you approved.}

\character{LIDA}
\dialogue{You may visit them in cold storage. You may not move in.}

\dialogue{Would she need a completion certificate?}

\action{She sees the saucer under the cabinet.}
\character{LIDA}
\dialogue{Is that yours?}
\character{CLARA}
\dialogue{No. It belongs here.}

\character{CLARA}
\dialogue{They say what went missing.}

\dialogue{That's not unusual, is it? People corrected things before computers.}
\character{CLARA}
\dialogue{Corrections were initialed.}

\character{JORY}
\dialogue{Is it your correction?}

\dialogue{No.}
\action{Clara opens the book wider, pressing on a spine she has spent her career protecting.}

\dialogue{Can I put it in with the rest?}
\action{The carton is labeled SCANNED -- DISPOSE. Clara takes the label off.}
\character{CLARA}
\dialogue{Not that one.}
