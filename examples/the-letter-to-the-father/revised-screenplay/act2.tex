% act2.tex — Movement II: The Office

\movement{II. THE OFFICE}

% ============================================================
% THE INSTITUTE — MORNING ARRIVAL
% ============================================================

\titlecard{WORKERS' ACCIDENT INSURANCE INSTITUTE \\ PRAGUE, 1912}

\scene{EXT. INSURANCE INSTITUTE BUILDING -- MORNING}

\action{A large, imposing building. Stone, columns, bronze doors already smudged by the morning rush of WORKERS and CLERKS.}

\action{Kafka arrives in a dark suit, briefcase in hand. He falls into the morning stream without breaking its pace.}

\scene{INT. INSURANCE INSTITUTE -- LOBBY -- CONTINUOUS}

\action{Marble floor. High ceilings. The sounds of an institution: footsteps, paper, the distant ringing of telephones. A large clock on the wall marks the hour.}

\action{Kafka moves through the lobby with practiced efficiency. He nods to a COLLEAGUE, checks his pigeonhole, removes a stack of files, and keeps moving.}

\cutto

% ============================================================
% THE ELEVATOR — OSCAR CLIP SCENE
% ============================================================

\scene{INT. INSURANCE INSTITUTE -- ELEVATOR LOBBY -- MORNING}

\action{A small lobby before a single elevator. Metal gate. An ELEVATOR OPERATOR, elderly, in a faded uniform, runs the mechanism.}

\action{Kafka waits. Three other MEN wait with him. They talk among themselves --- weather, football, the new regulation on form 17-B. Kafka stands slightly apart, holding his files.}

\action{The elevator arrives. The operator opens the gate. The three men step in.}

\action{The elevator is small. There is room for one more person. Kafka steps forward.}

\action{The operator clocks Kafka, the files in his arms, then the men already inside. He slides the gate toward shut.}

\action{The operator begins to close the gate.}

\character{KAFKA}
\dialogue{I can---}

\action{The gate closes. A clang of metal. The elevator begins to rise.}

\action{Kafka stands alone in the lobby, the files held against his chest. The numbers above the gate tick upward: 1... 2... 3...}

\action{No one in the corridor looks at him.}

\action{The clock ticks. Somewhere deep in the building, a stamp comes down hard on paper.}

\action{Kafka shifts the files to his other arm. The top page bends under his thumb.}

\action{He steps back from the gate and waits.}

\action{The elevator descends. The numbers tick down: 3... 2... 1. The gate opens. The operator looks at him.}

\character{ELEVATOR OPERATOR}
\dialogue{Going up?}

\action{Kafka steps in. The gate closes. The elevator rises. His reflection in the polished metal shows a thin man in a dark suit holding files at his chest.}

\cutto

% ============================================================
% KAFKA'S OFFICE — THE WORK
% ============================================================

\scene{INT. INSURANCE INSTITUTE -- KAFKA'S OFFICE -- DAY}

\action{A small, tidy office. Desk, files, a window overlooking a courtyard.}

\action{Kafka works. He reads a file. It concerns a FACTORY WORKER who lost two fingers in a stamping press. The file includes a medical report, a factory inspection form, a claim for compensation.}

\action{He reads carefully. Makes notes in the margin. His handwriting is small and precise.}

\action{A knock. DIRECTOR MARSCHNER, 55, enters in a solid suit, carrying the Kraus file and his hat, which he sets carefully on the cabinet before he speaks.}

\character{MARSCHNER}
\dialogue{Kafka. The Kraus file.}

\character{KAFKA}
\dialogue{The sawmill.}

\character{MARSCHNER}
\dialogue{They're contesting it. Saying the worker was negligent. That he positioned his hand---}

\character{KAFKA}
\dialogue{The guard was removed. I inspected the machine personally last month. The cylinder guard had been taken off because it slowed the feed rate by four seconds per board.}

\action{Marschner looks at him.}

\character{KAFKA}
\dialogue{Four seconds per board. Two hundred boards per shift.\\
Three shifts. Forty minutes a day.\\
Someone chose that over his hand.}

\character{MARSCHNER}
\dialogue{You inspected it personally?}

\character{KAFKA}
\dialogue{I inspect them all personally when I can. The forms are unreliable. The owners fill them in from memory. Or imagination.}

\action{Marschner sits down, one hand briefly on his knee as he settles. Studies Kafka.}

\character{MARSCHNER}
\dialogue{Sometimes I think you're the only person in this building who actually reads the files.}

\character{KAFKA}
\paren{a thin smile}
\dialogue{Someone has to.}

\character{MARSCHNER}
\dialogue{The recommendation?}

\character{KAFKA}
\dialogue{Full compensation. Inspect Kraus sawmills within thirty days.\\
I'll draft the order.}

\character{MARSCHNER}
\dialogue{Good. Good man.}

\action{He leaves. Kafka returns to the file. Opens it again. Looks at the medical photograph clipped to the inside: a man's hand, or what remains of it. Two stumps where fingers were.}

\action{Kafka studies the photograph until the paper clip marks his thumb. Then he closes the file, sets it in the outbox, and opens the next one.}

\action{The clock on the wall ticks.}

\cutto

% ============================================================
% THE WAITING ROOM
% ============================================================

\scene{INT. INSURANCE INSTITUTE -- WAITING ROOM -- DAY}

\action{A long room with wooden benches. WORKERS wait with caps in their hands. Bandaged wrists. Arms in slings. One man remains standing because sitting hurts his back.}

\action{Claim forms rest on their laps. A woman traces each printed line with her finger while her husband mouths the words beside her.}

\action{Kafka walks through the waiting room on his way to another office. He slows. Stops. Watches.}

\action{A YOUNG WORKER, barely 20, sits with his bandaged right hand resting on his knee. He tries to turn a page with his left hand and cannot manage it cleanly.}

\action{Kafka crosses to him and crouches beside the bench.}

\character{KAFKA}
\dialogue{May I?}

\action{The worker hands him the form. Kafka sees the printed word `NEGLIGENCE' and the blank line beneath it.}

\character{YOUNG WORKER}
\dialogue{The clerk said that line matters most.}

\character{KAFKA}
\dialogue{It does. Tell me exactly.}

\character{YOUNG WORKER}
\dialogue{The guard shook. The belt jumped. I reached in before it took the sleeve.}

\character{KAFKA}
\dialogue{Who buttons your collar now?}

\character{YOUNG WORKER}
\dialogue{My sister. Why?}

\action{Kafka turns the paper toward him and places the pencil in his left hand. He steadies the page with two fingers.}

\character{KAFKA}
\dialogue{Write: `The guard shook loose. The belt jumped.'}

\action{The worker writes slowly, awkwardly.}

\character{YOUNG WORKER}
\dialogue{Will it matter?}

\character{KAFKA}
\dialogue{More than `negligence' will.}

\character{KAFKA}
\dialogue{Then add: `Since the accident my sister buttons my collar.'}

\character{YOUNG WORKER}
\dialogue{That's not the machine. It's mornings.}

\character{KAFKA}
\dialogue{Exactly.}

\action{Kafka rises. The worker bends over the page and keeps writing as Kafka moves on.}

\cutto

% ============================================================
% MAX BROD — FIRST MEETING
% ============================================================

\scene{INT. CAFÉ ARCO -- EVENING}

\action{A Prague literary café. Loud, smoky, full of argument. WRITERS, INTELLECTUALS, and STUDENTS debate at crowded tables.}

\action{MAX BROD, 28, is already half out of his chair before Kafka reaches the table. Kafka's marked-up manuscript lies beside the coffee cups and a plate with two rolls Brod has clearly ordered for both of them.}

\action{Kafka approaches with a cup of coffee.}

\character{BROD}
\paren{standing, already half out of his chair}
\dialogue{Franz! Sit, sit. I missed my tram stop. I missed it completely. Because of you.}

\action{Kafka sits. The café noise swirls around them.}

\action{Brod pushes the plate a little closer to him. Kafka notices and declines without touching it.}

\character{KAFKA}
\dialogue{Then the railway has my apologies.}

\character{BROD}
\dialogue{Don't do that. Don't be modest. It's alive, this thing.}

\character{KAFKA}
\dialogue{I'm not a genius. I just don't sleep.}

\character{BROD}
\dialogue{``The Judgment.'' Franz. The opening --- I read it twice. I read it three times. The opening alone---}

\character{KAFKA}
\paren{uncomfortable}
\dialogue{It came out in one night. That's all I know about it.}

\character{BROD}
\dialogue{Good. Good. If you knew, you'd have improved it to death by morning.}

\action{Kafka reaches for his coffee instead of the pages. He stirs it once, twice, not looking at Brod.}

\character{KAFKA}
\dialogue{My father would say I should spend the night sleeping instead of writing.}

\character{BROD}
\dialogue{Your father sells buttons.}

\character{KAFKA}
\paren{sharply}
\dialogue{Fancy goods.}

\action{A beat. Then they both smile despite themselves.}

\action{Brod taps the manuscript, then produces an addressed envelope from his coat. Kafka sees it and goes still.}

\character{BROD}
\dialogue{I want to send it out.}

\character{KAFKA}
\dialogue{No.}

\character{BROD}
\dialogue{You brought it to me.}

\character{KAFKA}
\dialogue{Because you read quickly.\\
Not because you should mail it to strangers.}

\character{BROD}
\dialogue{Give it a week and you'll sand the skin off it.}

\action{Kafka's fingers go to the envelope, then stop on the table.}

\character{KAFKA}
\dialogue{Send it to whoever you like. I reserve the right to be horrified.}

\character{BROD}
\paren{beaming}
\dialogue{Done. Start with the roll. Then more coffee.}

\action{He wraps the untouched roll in a paper napkin and slides it beneath the corner of Franz's manuscript.}

\cutto

% ============================================================
% THE DOUBLE LIFE — OFFICE AND DESK
% ============================================================

\scene{INT. INSURANCE INSTITUTE -- KAFKA'S OFFICE -- DAY}

\action{Kafka at his desk. Files. Forms. Stamp. Signature. Date. His pen moves through official language without pause.}

\scene{INT. KAFKA'S BEDROOM -- NIGHT}

\action{Same hand. Same pen. A notebook instead of a file.}

\action{He writes in a notebook. The handwriting is smaller now, more urgent. He pauses. Crosses out a line. Writes again. The lamp burns low.}

\scene{INT. INSURANCE INSTITUTE -- KAFKA'S OFFICE -- DAY}

\action{A COLLEAGUE deposits a new stack of files on Kafka's desk. Kafka nods. Opens the top file.}

\action{Inside: a photograph of a machine. Belt, teeth, guard plate, grease-pencil marks in the margin.}

\scene{INT. KAFKA'S BEDROOM -- NIGHT}

\action{Kafka writes. Faster now. He reaches the edge of the page, goes back, strikes a word, keeps going.}

\vo{KAFKA}{As Gregor Samsa awoke one morning from uneasy dreams, he found himself transformed in his bed into a---}

\action{He stops with the nib hovering over the page. Looks at what he has written.}

\action{He writes the word.}

\cutto

% ============================================================
% THE FACTORY INSPECTION
% ============================================================

\scene{INT. TEXTILE FACTORY -- DAY}

\action{Noise. Machines. WORKERS at looms and presses. Kafka moves through the factory floor with a clipboard, accompanied by a reluctant FACTORY FOREMAN.}

\action{The machines are enormous and loud. Workers feed material into them without taking their eyes off the moving teeth.}

\character{FOREMAN}
\paren{shouting over the noise}
\dialogue{Everything is in order, as you can see. All guards in place. All regulations followed.}

\action{Kafka stops at a press. Examines the guard mechanism. It is loose. One bolt is missing entirely. He makes a note on his clipboard.}

\character{FOREMAN}
\dialogue{That's nothing. We've ordered the replacement.}

\character{KAFKA}
\dialogue{When?}

\character{FOREMAN}
\dialogue{Recently.}

\character{KAFKA}
\dialogue{The incident report from February says the same thing. ``Replacement ordered.'' It's now September.}

\action{The foreman's expression hardens.}

\character{FOREMAN}
\dialogue{Dr. Kafka, with respect, you're a lawyer. You sit in an office. You don't understand the realities of---}

\character{KAFKA}
\dialogue{I understand the gap between this guard and the press bed is forty-two millimeters.\\
A human finger is approximately eighteen millimeters in diameter.\\
I understand your workers are asked to bet, every hour of every shift, that their attention will never waver.}

\action{A pause. The machines roar around them.}

\character{KAFKA}
\dialogue{I'll note the bolt has not been replaced. The report will be filed by Friday.}

\action{He moves on. The foreman watches him go.}

\action{A WORKER at a nearby station --- a woman, middle-aged, hands quick and careful --- glances at Kafka as he passes. He glances back. Then she turns to the loom before the thread can slip.}

\scene{INT. INSURANCE INSTITUTE -- KAFKA'S OFFICE -- LATER}

\action{Kafka at his desk. He opens the Kraus file. Inside: the medical photograph. A worker's hand. Two stumps where fingers were.}

\action{He studies it. Then closes the file and reaches for the inspection form.}

\cutto

% ============================================================
% THE PROMOTION
% ============================================================

\scene{INT. INSURANCE INSTITUTE -- MARSCHNER'S OFFICE -- DAY}

\action{A larger office. Leather chairs. A window with a view. Marschner behind his desk, collar loosened, Board memoranda spread over a plate holding half an untouched sandwich. Kafka stands.}

\character{MARSCHNER}
\dialogue{The Kraus report was exemplary. Your inspection findings forced a settlement in two days. I'm recommending you for senior assessor. Larger office, supervisory responsibility, a raise.}

\action{Kafka's eyes go first to the corner window, then to the half sandwich gone cold on Marschner's plate.}

\character{MARSCHNER}
\dialogue{Office twenty-seven. Corner window. Door that shuts. One clerk who will fear you a little.}

\action{Marschner opens a drawer, takes out a brass key on a paper tag marked `27,' and sets it on the desk between them.}

\character{MARSCHNER}
\dialogue{Take it. See how it feels in your pocket.}

\action{Kafka picks up the key. It is small, ordinary, heavier than it looks.}

\character{KAFKA}
\dialogue{Heavier than the files.}

\character{MARSCHNER}
\dialogue{It gets heavier. Then you stop noticing.}

\character{KAFKA}
\dialogue{I appreciate your confidence, Director.}

\character{MARSCHNER}
\dialogue{You've earned it.}

\character{KAFKA}
\dialogue{I'd like to decline.}

\action{Marschner stares at him.}

\character{MARSCHNER}
\dialogue{Decline.}

\character{KAFKA}
\dialogue{The additional hours. The meetings. My current position leaves me a part of the day that is still mine.}

\character{MARSCHNER}
\dialogue{Time for what? You never leave before dusk as it is.}

\character{MARSCHNER}
\dialogue{If you've found a part of the day this building can't take, keep it.}

\character{MARSCHNER}
\dialogue{You're the most competent man in this building and you want to stay in a box office reading claim files.}

\character{KAFKA}
\paren{a faint smile}
\dialogue{The files are honest. A machine takes a hand and says so plainly. I find that restful.}

\character{KAFKA}
\dialogue{If I take the key, I will still be here at midnight.}

\action{Marschner shakes his head and lets the document fall closed on his desk. He glances once at the untouched sandwich, then back at Kafka.}

\action{Kafka sets the key back on the desk. It lies between them. Marschner's hand stays on the closed file. Kafka's stays at his side.}

\character{MARSCHNER}
\dialogue{The offer stands. If you change your mind.}

\action{Kafka nods and leaves. In the corridor, he exhales only when Marschner's door shuts behind him.}

\cutto

% ============================================================
% BROD AND KAFKA — THE WALK
% ============================================================

\scene{EXT. PRAGUE -- CHARLES BRIDGE -- LATE AFTERNOON}

\action{Kafka and Brod walk. The bridge is beautiful in the failing light. The saints on their pedestals look down with stone patience.}

\character{BROD}
\dialogue{You refuse promotion. You refuse publishers. You refuse everything. Do you mean to disappoint both careers equally, or is one getting a head start?}

\character{KAFKA}
\dialogue{It saves time.}

\character{BROD}
\dialogue{You have three stories --- three --- as good as anything being written in German. In any language. The best short prose in Europe is sitting in your desk drawer.}

\character{KAFKA}
\dialogue{They're sitting in my desk drawer because they're not finished.}

\character{BROD}
\dialogue{They are as finished as you will ever let them be. And once they're written --- Franz, listen to me --- once they're written, they don't belong to you. They belong to whoever reads them.}

\character{KAFKA}
\dialogue{That's the terrifying part.}

\character{BROD}
\dialogue{That's the point of literature.}

\character{KAFKA}
\dialogue{Writing is one thing.\\
Handing it to strangers is another.}

\action{They keep walking. Kafka trails two fingers along the cold stone parapet. Below them, the river works under the arches in the failing light.}

\character{KAFKA}
\dialogue{All day at the Institute I wait for night. Then night comes and I miss the office.}

\character{BROD}
\dialogue{That isn't an argument against writing.}

\character{KAFKA}
\dialogue{No. It's an argument against me.}

\action{Kafka looks at him, then away to the river.}

\character{KAFKA}
\dialogue{You're a good friend.}

\character{BROD}
\dialogue{I know. It's what I'm good for.}

\character{KAFKA}
\dialogue{It's not a small one.}

\action{They walk. The light fades. The saints on the bridge darken into silhouettes.}

\cutto

% ============================================================
% THE CORRIDOR — REALITY SLIPS
% ============================================================

\scene{INT. INSURANCE INSTITUTE -- CORRIDOR -- LATE AFTERNOON}

\action{End of the day. The building is emptying. Kafka walks down a long corridor toward the exit. His footsteps echo on the stone floor.}

\action{The corridor is ordinary. Doors on either side. Small brass plates with numbers.}

\action{He passes door 14. Door 15. Door 16. He should already be at the stairs.}

\action{The next brass plate reads KR-1742/11. The next: W-803/9. Claim numbers.}

\action{Kafka stops. Somewhere out of sight: a typewriter carriage snaps, then a saw catches on wood, then silence.}

\action{He looks behind him. The corridor stretches back, identical. No stairs.}

\action{He turns forward and walks faster. More doors. More numbers. One brass plate reads: ALTSTÄDTER RING 6 --- his father's shop address. Then it reads KR-2201/4 again.}

\action{Then: the exit. Glass doors. Streetlight through the panes. A tram bell outside.}

\action{Kafka pushes through the doors into the evening air.}

\action{He stands on the steps and looks back through the glass. Inside: a short corridor ending where it should.}

\action{He adjusts his hat. Walks down the steps. Goes home.}
