% act4.tex — Movement IV: The Illness

\movement{IV. THE ILLNESS}

% ============================================================
% THE DIAGNOSIS
% ============================================================

\titlecard{PRAGUE, AUGUST 1917}

\scene{INT. KAFKA'S BEDROOM -- NIGHT}

\action{Three in the morning. Kafka wakes coughing. He sits up in the narrow bed, hand pressed to his mouth.}

\action{When the coughing stops, he looks at his hand. The moonlight through the window shows the dark stain clearly.}

\action{He turns on the lamp. Looks at his hand more carefully. It is unmistakable.}

\action{He gets up. Goes to the washstand. Rinses his hand. Rinses it again. The water in the basin turns pink.}

\action{He stands there for a long time. Then he does something unexpected: he smiles. A strange, private smile --- the smile of a man who has just received confirmation of something he has always suspected.}

\cutto

\scene{INT. DOCTOR'S OFFICE -- PRAGUE -- DAY}

\action{A consulting room. Clean. Clinical. DR. MÜHLSTEIN, 50s, a careful man with careful hands, sits across from Kafka. Between them, on the desk: an X-ray plate. The white shadow on the lung is clearly visible.}

\character{DR. MÜHLSTEIN}
\dialogue{The left lung, primarily. There is involvement of the bronchial tissue. The hemorrhage you described is consistent with---}

\character{KAFKA}
\dialogue{Tuberculosis.}

\character{DR. MÜHLSTEIN}
\dialogue{Yes.}

\action{A pause. The doctor waits for the reaction --- the shock, the denial, the questions. Kafka provides none of these.}

\character{KAFKA}
\dialogue{What do you recommend?}

\character{DR. MÜHLSTEIN}
\dialogue{Rest. Mountain air, if possible. A sanatorium. Your diet must be carefully managed. And your work---}

\character{KAFKA}
\dialogue{My work at the Institute?}

\character{DR. MÜHLSTEIN}
\dialogue{Must be reduced. Ideally suspended. Your body needs---}

\character{KAFKA}
\dialogue{My body and I have never been on good terms. I don't expect a reconciliation now.}

\action{The doctor studies him.}

\character{DR. MÜHLSTEIN}
\dialogue{Herr Kafka. This is serious.}

\character{KAFKA}
\paren{quietly, almost gently}
\dialogue{I know, Doctor. I only thought it would stay on the page.}

\cutto

% ============================================================
% TELLING OTTLA
% ============================================================

\scene{EXT. PRAGUE -- SMALL PARK -- AFTERNOON}

\action{Kafka and Ottla on a bench. Autumn. Leaves falling with slow deliberation.}

\character{KAFKA}
\dialogue{Tuberculosis. The left lung.}

\action{Ottla is silent. She takes his arm. Pulls him close. He lets her.}

\character{OTTLA}
\dialogue{You're coming to Zürau. The cottage. Fresh air. No one to bother you.}

\character{KAFKA}
\dialogue{Father will say I'm being dramatic.}

\character{OTTLA}
\dialogue{He can say whatever he likes.}

\action{Her tone is final. She is, in this moment, the strongest person in Kafka's life.}

\cutto

% ============================================================
% ZÜRAU — THE COUNTRY
% ============================================================

\scene{EXT. ZÜRAU -- VILLAGE -- DAY}

\action{A small village in northwest Bohemia. Fields. Low stone walls. The extraordinary quiet of a place where the principal sounds are wind and animal.}

\action{Kafka walks a dirt path. He is wearing simpler clothes than Prague requires. He breathes the country air with the careful attention of a man who has been told his breathing now has administrative significance.}

\scene{INT. OTTLA'S COTTAGE -- ZÜRAU -- DAY}

\action{A simple room. A bed, a desk, a window facing the fields. Ottla has made it clean and warm. Kafka sets down his bag.}

\action{Ottla comes in after him carrying kindling under one arm, her skirt hem muddy, her sleeves rolled. This refuge has not appeared by miracle.}

\action{He goes to the desk. Touches it. Sits in the chair.}

\action{Through the window: fields, sky, the slow movement of clouds. No insurance forms. No father's voice. No Prague.}

\action{He opens his notebook. Writes.}

\vo{KAFKA}{A cage went in search of a bird.}

\action{He writes. In Zürau, something happens to his prose. It becomes shorter. Sharper. The aphorisms he produces here are like stones --- small, hard, irrefutable.}

\scene{INT. OTTLA'S COTTAGE -- ZÜRAU -- EVENING}

\action{Kafka and Ottla eat a simple dinner. Bread. Cheese. Milk. Ottla cuts the cheese into two pieces, gives him the larger one, and pretends not to notice each time he stops chewing. When he pauses too long, she nudges the milk toward him without comment.}

\character{OTTLA}
\dialogue{You're eating.}

\character{KAFKA}
\dialogue{I'm negotiating with the bread. It's a different thing.}

\character{OTTLA}
\dialogue{The country air suits you.}

\character{KAFKA}
\dialogue{The quiet suits me. Let the air take the credit.}

\character{OTTLA}
\dialogue{The air and I milked two cows before dawn, so I'd like equal credit.}

\action{She looks at him. Directly.}

\character{OTTLA}
\dialogue{You know he asks about you. Every day. He asks Mother to read your letters aloud at dinner.}

\character{KAFKA}
\dialogue{That sounds right. He likes me better on paper.}

\action{Ottla shakes her head, but one corner of her mouth gives way. She tears off another piece of bread and sets it on his plate without looking at him.}

\cutto

% ============================================================
% THE FATHER VISITS
% ============================================================

\scene{EXT. ZÜRAU -- VILLAGE ROAD -- AFTERNOON}

\action{A hired car on the dirt road. Out of place among the fields. HERMANN KAFKA climbs out, overcoat, city shoes, carrying a large wicker basket heavy enough to shift his weight.}

\action{He looks at the village. The mud. The sky. The absolute absence of commerce.}

\scene{INT. OTTLA'S COTTAGE -- ZÜRAU -- CONTINUOUS}

\action{Kafka sits at the desk, notebook open. A knock. Ottla answers. Her face changes.}

\action{Hermann fills the doorframe. He sets the basket on the table. It lands with authority.}

\character{HERMANN}
\dialogue{Your mother packed it. Smoked meat. That cheese you used to eat.\\
Real bread, not whatever the country calls bread.}

\action{He looks at his son. At the narrow bed, the bare walls, the desk with its notebooks. At Kafka, who is thinner than the last time they stood in the same room.}

\character{HERMANN}
\dialogue{You look---}

\action{He stops himself. Starts again.}

\character{HERMANN}
\dialogue{The air is supposed to help.}

\character{KAFKA}
\dialogue{It does help.}

\character{HERMANN}
\dialogue{Good. Eat the cheese. Your mother asked three times if I had the cheese.}

\action{He sits in the only other chair. It is too small for him. His knees nearly touch the desk. The room, which was enough for Kafka, contracts around Hermann the way all rooms do.}

\action{Silence. Hermann looks at the notebooks on the desk. Does not ask. Looks out the window at the fields instead.}

\character{HERMANN}
\dialogue{What do you do here all day?}

\character{KAFKA}
\dialogue{I rest. I walk. I write a little.}

\character{HERMANN}
\dialogue{You could rest in Prague.\\
Your mother would cook for you properly. Not---}

\action{He gestures at the cottage. At all of it.}

\character{KAFKA}
\dialogue{The quiet helps, Father.}

\character{HERMANN}
\dialogue{Quiet. In Prague there is quiet. Close the door.}

\action{Kafka says nothing. Hermann opens the basket. Unwraps the cheese. Sets it on the table. Unwraps the bread. Cuts two thick slices and pushes one toward his son.}

\character{HERMANN}
\dialogue{Eat.}

\action{Kafka picks up the bread. Takes a small bite. Hermann watches until it is swallowed, then eats his own slice in three bites.}

\action{For a moment they are just a father and a son eating bread in a small room. It is almost ordinary. It is almost enough.}

\character{HERMANN}
\dialogue{You need weight. The doctor said weight.}

\character{KAFKA}
\dialogue{Thank you for coming.}

\action{Hermann looks at him. Something passes across his face --- not tenderness, but the shadow of what tenderness would look like in a man who has never had the word for it.}

\character{HERMANN}
\dialogue{Your mother worries.}

\action{He stands. The chair gives back its space.}

\character{HERMANN}
\dialogue{Eat the cheese, Franz. The whole thing. Don't write about it, eat it.}

\action{He puts on his hat. Stops at the door. Turns back.}

\character{HERMANN}
\dialogue{The shop is fine. In case you were wondering.}

\character{KAFKA}
\dialogue{I was wondering.}

\action{Hermann leaves. The hired car starts. Kafka watches it until the road dust settles.}

\action{On the table: the bread, the cheese, the smoked meat. The basket. The notebooks pushed to one side to make room for it.}

\action{Ottla comes in from the back. Looks at the food. Looks at her brother.}

\character{OTTLA}
\dialogue{He came all this way?}

\character{KAFKA}
\dialogue{He brought cheese.}

\action{Ottla picks up the knife and cuts the cheese properly. Neither of them says what the visit meant. Neither of them needs to.}

\cutto

% ============================================================
% THE FILE FINDS HIM
% ============================================================

\scene{INT. OTTLA'S COTTAGE -- ZÜRAU -- AFTERNOON}

\action{Rain against the window. Ottla comes in with the post and a pair of muddy boots in her hand. One envelope bears the seal of the Workers' Accident Insurance Institute.}

\character{OTTLA}
\dialogue{Your office can track livestock now.}

\action{Kafka takes the envelope. Opens it. Inside: a short note from Marschner and a thin file tied with string. KRAUS SAWMILLS.}

\action{INSERT: ``Kafka --- Kraus has appealed the compensation order and the mandatory inspections. The Board wants compromise. I would prefer accuracy. If you have strength for a memorandum, I need it. Not only on Kraus. On guard standards generally. If we can make one owner's fraud read like an administrative problem, we may force them to fix more than one mill. Marschner.''}

\action{Kafka reads. Coughs into his handkerchief. Reads again.}

\character{OTTLA}
\dialogue{You don't have to answer that.}

\character{KAFKA}
\dialogue{I know. That's why I will.}

\action{He sits at the desk. Opens the file. Inspection notes. A worker's statement. A factory owner's letter explaining that every precaution had been taken. He lays the worker's statement on top.}

\action{Kafka writes. Coughs into the handkerchief. Waits. Keeps writing. One adjective gets crossed out. A measurement replaces it.}

\action{He finishes the memorandum on Kraus. Pulls a second clean page toward him. At the top he writes: ``Minimum Guard Requirements for Circular Saws and Feed Cylinders.''}

\action{He numbers the hazards one by one. Missing guards. Feed speed. Distance of hands from blade. By the time he stops, Kraus's name appears only twice. He squares the pages and sets them aside.}

\action{Then he pulls a fresh stack of paper toward him. There is another case, older and unwinnable, for which no office exists.}

\cutto

% ============================================================
% THE LETTER TO HIS FATHER — CENTERPIECE
% ============================================================

\titlecard{SCHELESEN, NOVEMBER 1919}

\scene{INT. PENSION STÜDL -- KAFKA'S ROOM -- NIGHT}

\action{A small room in a pension in the countryside. Kafka at the desk. Written pages to his left. Blank paper to his right. The written pile is already thick.}

\vo{KAFKA}{Dearest Father --- You asked me recently why I maintain\\
that I am afraid of you. As usual, I was unable\\
to think of any answer, partly because the basis\\
of my fear is so great that it goes beyond\\
my power of speech and even my power of thought.}

\action{He writes without the usual delay. Dips the pen. Turns the page. Keeps going.}

\scene{INT. PENSION STÜDL -- KAFKA'S ROOM -- LATER}

\action{Thirty pages. Forty. The lamp burns lower. A glass of water. A cup of coffee gone cold.}

\vo{KAFKA}{Sometimes I imagine the map of the world spread out and you stretched diagonally across it. And I feel as if I could consider living in only those regions that either are not covered by you or are not within your reach. And in keeping with the image I have of your magnitude, these regions are not many and not very comforting.}

\action{He drinks. His hand trembles once. He waits for it to steady, then keeps writing.}

\scene{INT. PENSION STÜDL -- KAFKA'S ROOM -- LATER STILL}

\action{Kafka writes. Outside, the village is dark and silent.}

\action{Fifty pages. Sixty. The handwriting stays small, exact. By dawn the stack has outgrown the blotter.}

\vo{KAFKA}{My writing was all about you; all I did there, after all, was to bemoan what I could not bemoan upon your breast.\\
It was an intentionally long-drawn-out leave-taking from you, yet, although enforced by you, it did take its course in the direction determined by me.}

\action{He sets down the pen. Flexes his hand. Turns the whole stack over and reads from the beginning. The pages make a dry, official sound in the room.}

\action{At the end he sits very still. Then he ties the bundle with string, wraps it, and writes on the front: ``For Father.''}

\action{After a beat, he adds: ``Via Mother.''}

\cutto

% ============================================================
% THE DOOR
% ============================================================

\scene{INT. KAFKA FAMILY APARTMENT -- HALLWAY -- AFTERNOON}

\action{Kafka stands in the hallway of the family apartment. The manuscript is in his hands, wrapped and tied. He faces his father's study door.}

\action{Through the door: the scratch of a pen on a ledger. The creak of Hermann's chair. A cough. The small domestic sounds of a man doing accounts.}

\action{Kafka's hand rises to the door handle. Touches it. The metal is cold.}

\action{He stands there. Ten seconds. Twenty. His hand on the handle. His father's breathing on the other side.}

\action{Inside the study, Hermann shifts his weight. The chair protests. A page turns.}

\action{Kafka's grip tightens. Then loosens. Then tightens again.}

\action{From the kitchen, the sound of Julie filling the kettle. The small mercy of another person in the apartment.}

\action{Kafka takes his hand off the handle.}

\action{He stands in the hallway a moment longer. Then he turns and walks toward the kitchen, the manuscript still in his hands.}

\cutto

% ============================================================
% THE LETTER — JULIE'S CHOICE
% ============================================================

\scene{INT. KAFKA FAMILY APARTMENT -- LIVING ROOM -- DAY}

\action{Kafka places the wrapped bundle in Julie's hands. She is startled by the weight.}

\character{JULIE}
\dialogue{What is this?}

\character{KAFKA}
\dialogue{A letter. Will you give it to Father?}

\action{Julie reads the front. Looks toward the bedroom door. Beyond it: Hermann snoring faintly in his afternoon sleep.}

\character{JULIE}
\dialogue{Franz...}

\character{KAFKA}
\dialogue{Please. He doesn't have to answer. I only need him to read it.}

\action{Julie sits with the bundle on her lap.}

\character{JULIE}
\paren{very quietly}
\dialogue{I can't give this to him, Franz.}

\character{KAFKA}
\dialogue{Mother---}

\character{JULIE}
\dialogue{I know what's in it without opening it.}

\character{JULIE}
\dialogue{If I carry this into that room, he will hear accusation before he hears pain. You will hear mockery before you hear anything at all. By supper the whole apartment will be sick with it.}

\character{KAFKA}
\dialogue{That's his problem.}

\character{JULIE}
\dialogue{No. That's all of us. That's what a family is.}

\action{Kafka stares at the bundle in her lap.}

\character{KAFKA}
\dialogue{You've carried things between us my whole life.}

\character{JULIE}
\dialogue{Exactly. Which is why I won't carry this one.}

\character{KAFKA}
\paren{very quietly}
\dialogue{Then it was all for nothing.}

\character{JULIE}
\dialogue{No. You finally said it. That is not nothing.}

\action{Kafka nods. He leaves the apartment.}

\action{Julie listens to his steps fade. Hermann snores in the next room. She opens her writing desk, places the bundle inside, and closes the drawer.}

\scene{INT. KAFKA FAMILY APARTMENT -- DINING ROOM -- EVENING}

\action{Supper. Hermann at the head of the table. Julie serving soup. Franz's place has been laid and cleared again so quickly that the damp ring from the lifted bowl still darkens the cloth.}

\character{HERMANN}
\dialogue{Any word from Franz?}

\character{JULIE}
\dialogue{He is resting. The country air helps.}

\character{HERMANN}
\dialogue{Good. Let it cure him of analysis. A man coughs once and suddenly the whole family becomes literature.}

\action{Julie sets down the bread basket. Says nothing.}

\character{HERMANN}
\dialogue{Did he send another packet? He always writes too much when a postcard would do.}

\character{JULIE}
\dialogue{Only a note.}

\action{Hermann eats. Julie watches the writing desk from across the room without appearing to.}

\character{HERMANN}
\dialogue{Tell him to come to supper when he has appetite enough to eat like a man.}

\action{Julie lifts the extra spoon from beside Franz's place setting and carries it back to the sideboard. Later. Hermann rises and crosses to the writing desk for the household ledger. Julie is up before he can touch the top drawer.}

\character{HERMANN}
\dialogue{Where is it?}

\character{JULIE}
\dialogue{Lower drawer. I moved it.}

\action{She opens the lower drawer, hands him the ledger, and closes it again. Hermann returns to the table, satisfied. Julie keeps her hand on the top drawer until he looks down at the ledger and not at her.}

\cutto

% ============================================================
% JULIE AND OTTLA — THE KITCHEN
% ============================================================

\scene{INT. KAFKA FAMILY APARTMENT -- KITCHEN -- MORNING}

\action{Early. Before Hermann wakes. Julie at the stove. Ottla enters with a paper bag from the market --- bread, eggs. She sets it on the counter without being asked.}

\action{Julie takes the eggs, cracks two into a pan. Ottla hangs her coat, rolls her sleeves, starts slicing the bread. They work around each other with the ease of women who have shared this kitchen for twenty years.}

\character{OTTLA}
\dialogue{You look tired.}

\character{JULIE}
\dialogue{I've looked tired since 1898. It's practically a feature.}

\action{Ottla almost smiles. She puts the kettle on.}

\character{OTTLA}
\dialogue{How is the new girl at the shop?}

\character{JULIE}
\dialogue{Hopeless with the ribbon but good with customers. Your father trains her the way he trains everyone --- volume and repetition.}

\character{OTTLA}
\dialogue{The two things he's never short of.}

\action{Julie slides an egg onto a plate and sets it in front of Ottla.}

\character{JULIE}
\dialogue{Franz left something with me. For your father. A letter.}

\character{OTTLA}
\dialogue{How long?}

\character{JULIE}
\dialogue{A hundred pages.}

\action{Ottla stops slicing.}

\character{OTTLA}
\dialogue{And you kept it.}

\character{JULIE}
\dialogue{I put it in the writing desk. Top drawer.}

\character{OTTLA}
\dialogue{Good.}

\character{JULIE}
\dialogue{Good?}

\character{OTTLA}
\dialogue{If you'd given it to Father, he'd have read ten pages, called it ingratitude, and used the rest to light the stove. You know that.}

\character{JULIE}
\dialogue{I know that.}

\action{A pause. Ottla pours two cups of coffee. Sets one in front of her mother.}

\character{JULIE}
\dialogue{Thirty years of carrying things between them. And he asks me to carry the one thing that would break.}

\character{OTTLA}
\dialogue{That's why he asked you. Because you'd know not to.}

\action{Julie looks at her daughter. Something passes between them --- not revelation, but recognition. Two women who have lived inside the same man's gravity and found different ways to stand.}

\character{JULIE}
\dialogue{When did you get so sharp?}

\character{OTTLA}
\dialogue{The farm. Animals don't explain. You watch what they do.}

\action{Julie laughs. A real laugh, quiet and surprised.}

\character{JULIE}
\dialogue{Eat your egg.}

\character{OTTLA}
\dialogue{You eat yours.}

\character{JULIE}
\dialogue{I'm not hungry.}

\character{OTTLA}
\dialogue{You and Franz.}

\action{That lands differently than Ottla intended. Julie picks up her fork.}

\action{From the bedroom: Hermann turning over. The bedsprings. The familiar sound of a large man deciding whether the day has earned his attention yet.}

\character{OTTLA}
\dialogue{I'll talk to him.}

\character{JULIE}
\dialogue{Be gentle.}

\character{OTTLA}
\dialogue{I'm always gentle. I just don't look it.}

\action{She takes her coat and goes. Julie finishes the egg. Washes both plates. The drawer with the letter is three steps from where she stands. She does not look at it.}

\cutto

% ============================================================
% OTTLA — AFTER THE LETTER
% ============================================================

\scene{EXT. PRAGUE -- VLTAVA RIVER EMBANKMENT -- AFTERNOON}

\action{Kafka and Ottla walk beside the river. Winter. Their breath hangs in the air.}

\character{OTTLA}
\dialogue{Mother told me about the letter. And that she won't deliver it.}

\action{Kafka says nothing.}

\character{KAFKA}
\dialogue{She was right not---}

\character{OTTLA}
\dialogue{---not to. I know. She was useful. That's the word you want.}

\character{OTTLA}
\dialogue{You wrote a hundred pages because paper doesn't interrupt. Then you gave them to Mother because she does.}

\character{KAFKA}
\dialogue{If I'd put it in his hand---}

\character{OTTLA}
\dialogue{He would have turned it into a scene. Yes. And you would have been in the scene. That's what you can't bear.}

\action{Kafka stops walking. Ottla stops with him.}

\character{OTTLA}
\dialogue{You stood at his door, Franz. I know because I know you. And you walked away and gave it to Mother instead.}

\character{KAFKA}
\paren{very quietly}
\dialogue{Cruelest thing you've said to me.}

\character{OTTLA}
\dialogue{I learned it from Father. I just say it to your face.}

\action{A long silence. The river moves.}

\character{KAFKA}
\dialogue{You're right.}

\character{OTTLA}
\dialogue{I know. I'm sorry.}

\action{She takes his arm. They walk. Two children of the same father, marked differently by the same wound.}

\cutto

% ============================================================
% BROD VISITS — THE DISAGREEMENT
% ============================================================

\titlecard{TATRANSKÉ MATLIARE, 1921}

\scene{INT. SANATORIUM -- KAFKA'S ROOM -- DAY}

\action{Brod sits across from Kafka. Between them, on the table: a plate of food Kafka has barely touched and two cups of tea. Kafka is thinner. The sanatorium has given him regularity but not recovery. Brod has already cut one roll into smaller pieces and lined them up by Kafka's hand.}

\action{Brod reaches for his tea, picks up Kafka's cup by mistake, puts it back, picks up the right one, puts that back too. His hands want a task they cannot find.}

\character{BROD}
\dialogue{You look --- well. You look well.}

\character{KAFKA}
\dialogue{I look like a man who has been told to eat six times a day and manages three.}

\character{BROD}
\dialogue{Three is better than two. Three is something.}

\character{KAFKA}
\dialogue{Max. Spare me.}

\action{Brod deflates. Then tears off the smallest piece of bread, sets it by Kafka's hand, and only then pulls a journal from his bag.}

\character{BROD}
\dialogue{I brought you --- here. The new issue. Page forty-three. Your story. It's extraordinary, Franz. The best thing in the---}

\action{Kafka takes the journal. Opens it. Finds his story. Reads the first line. His expression does something complicated --- pride and embarrassment and something like grief, all at once.}

\character{KAFKA}
\dialogue{They misspelled ``Josefine.''}

\character{BROD}
\dialogue{No one will care.}

\character{KAFKA}
\dialogue{I care.}

\character{BROD}
\dialogue{Of course you care. You're right. You're always right about this. Eat one bite and continue being right.}

\action{Kafka looks at him, irritated and grateful in the same glance. He takes the piece, eats it, swallows.}

\action{Brod's face changes. Serious now.}

\character{BROD}
\dialogue{Franz. I need to ask you something.}

\character{KAFKA}
\dialogue{About the manuscripts.}

\character{BROD}
\dialogue{You have drawers full of unpublished work. Novels. Stories. Fragments. If anything were to---}

\character{KAFKA}
\dialogue{If I were to die, you mean.}

\character{BROD}
\dialogue{I didn't say that.}

\character{KAFKA}
\dialogue{You were about to say it. Doctors say ``options'' when they mean they are running out of them.}

\action{Brod is quiet.}

\character{KAFKA}
\dialogue{I've thought about it. I want them destroyed.}

\action{Brod stares at him.}

\character{BROD}
\dialogue{You can't mean that.}

\character{KAFKA}
\dialogue{They're unfinished. Unpublishable. Why would I want that kept?}

\character{BROD}
\dialogue{Because it matters. Even the fragments do.\\
It belongs to---}

\character{KAFKA}
\dialogue{Don't say ``the world.'' The world didn't write them. I did. And I didn't finish them.}

\character{BROD}
\dialogue{Literature is not a house. A half-finished novel by you is worth more than the complete works of---}

\character{KAFKA}
\dialogue{Now you're flattering me. You've lost the argument.}

\action{A pause. The tension between them is real --- not hostility, but the friction of two men who love each other and disagree about something fundamental. Brod believes the work belongs to humanity. Kafka believes the work belongs to the fire.}

\character{BROD}
\dialogue{I'll respect your wishes.}

\character{KAFKA}
\dialogue{You say that the way I say ``I'll sleep earlier.'' Neither of us means it.}

\action{Brod reaches across the table and puts his hand on Kafka's. A gesture of friendship so plain and unadorned that Kafka does not know how to receive it. He sits with Brod's hand on his and looks out the window at the mountains and says nothing.}

\cutto

% ============================================================
% THE BODY FAILS — MONTAGE
% ============================================================

\scene{INT. SANATORIUM -- VARIOUS -- DAY / NIGHT}

\action{A sequence. Clinical, rhythmic, like the stations of an administrative process:}

\action{Kafka on a scale. The NURSE records the number. It is lower than last week.}

\action{Kafka at a meal table. Soup, bread, milk. He eats the bread. Sips the milk. Pushes the soup away.}

\action{Kafka in a chair on a terrace, wrapped in blankets. Mountain air. Other PATIENTS nearby, similarly wrapped. No one speaks.}

\action{Kafka coughing. Into a handkerchief. He folds it without looking at it.}

\action{A letter arrives. Institute stationery. Kafka opens it at the terrace table.}

\action{INSERT: ``Kafka --- the young man from the waiting room, the belt-press case. You will be glad to know his claim was approved in full. He is working again, left-handed, at a different factory. Your wording on the form made the difference. --- Marschner.''}

\action{Kafka reads it. Folds it once. Reads it again.}

\action{He picks up his notebook. The handwriting is the same. The hand that writes it is thinner.}

\cutto

% ============================================================
% THE CORRIDOR — REALITY SHIFTS
% ============================================================

\scene{INT. SANATORIUM -- CORRIDOR -- NIGHT}
