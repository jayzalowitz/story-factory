% act6.tex — Movement VI: The Last Room

\movement{VI. THE LAST ROOM}

% ============================================================
% KIERLING SANATORIUM
% ============================================================

\titlecard{KIERLING, NEAR VIENNA \\ APRIL 1924}

\scene{EXT. SANATORIUM KIERLING -- DAY}

\action{A low building among trees. Austrian countryside. Spring. The cruel irony of flowers.}

\action{An AMBULANCE arrives. Kafka is helped out --- he can walk, but barely. Dora is with him. ROBERT KLOPSTOCK, 24, a young Hungarian medical student who has become Kafka's friend and caretaker, supports him on the other side.}

\action{Kafka looks at the building. White walls. Windows. A door.}

\character{KAFKA}
\paren{to Klopstock, hoarse}
\dialogue{It's smaller than the last one.}

\character{KLOPSTOCK}
\dialogue{The rooms are good. The doctor is good.}

\character{KAFKA}
\dialogue{You keep saying ``good.'' That worries me.}

\action{Klopstock almost smiles. Dora does not.}

\cutto

% ============================================================
% THE ROOM
% ============================================================

\scene{INT. SANATORIUM KIERLING -- KAFKA'S ROOM -- DAY}

\action{A private room. A bed. A window overlooking the garden. A small table beside the bed. A water glass. A bell to call the nurse.}

\action{Kafka is settled into the bed. He is propped on pillows. His neck is swollen --- the tuberculosis has reached his larynx. Speaking is painful. Eating is worse.}

\action{DR. HOFFMANN, 50s, enters. He pauses a fraction before approaching the bed.}

\character{DR. HOFFMANN}
\dialogue{Herr Kafka. Welcome. We'll make you as comfortable as possible.}

\action{Kafka nods. He does not spend words.}

\action{On the bedside table: his notebook, a pencil, a glass of water.}

\action{Kafka picks up the pencil. Writes on a small slip of paper. Shows it to the doctor.}

\action{INSERT: The slip reads: ``How long do flowers usually last in a vase? I ask for planning purposes.''}

\action{The doctor reads it. Looks at Kafka. Kafka's eyes, above the swollen throat, are clear, dark, alive with a humor that refuses to stop.}

\cutto

% ============================================================
% THE SLIPS OF PAPER
% ============================================================

\scene{INT. SANATORIUM KIERLING -- KAFKA'S ROOM -- DAY}

\action{Days pass. The room becomes a world of written notes.}

\action{Kafka communicates by slips of paper. He writes them constantly --- to Dora, to Klopstock, to the nurses, to the doctor. The slips accumulate on the bedside table like fallen leaves.}

\action{A NURSE brings a tray. Soft food. Broth. A glass of fruit juice.}

\action{Kafka looks at the tray. He picks up his pencil. Writes.}

\action{INSERT: ``The fruit juice is very beautiful to look at. That is the highest compliment I am able to pay it.''}

\action{The nurse reads it. She does not know whether to smile or cry. She settles for something between the two.}

\action{Another slip, to Dora:}

\action{INSERT: ``A little water. The lilac --- could you move it closer? But not too close. Let me admire it from a politician's distance.''}

\action{Dora moves the flowers. Her hands are steady. Her face is not.}

\action{Another slip, to Klopstock:}

\action{INSERT: ``Do you think the lemonade is angry with me? I keep looking at it and not drinking it. That must be hurtful for a lemonade.''}

\action{Klopstock reads it. Laughs. Kafka watches him laugh and his eyes smile even though his mouth cannot.}

\cutto

% ============================================================
% A HUNGER ARTIST — THE PROOFS
% ============================================================

\scene{INT. SANATORIUM KIERLING -- KAFKA'S ROOM -- DAY}

\action{Kafka sits propped in bed. On his lap: galley proofs for ``A Hunger Artist.'' On the bedside tray: broth, bread, fruit juice, all barely touched.}

\action{He reads with ferocious attention, pencil in hand, correcting commas, restoring a word, refusing sloppiness even here.}

\action{Kafka reads to the final lines and stops.}

\action{From the corridor: the rattle of a lunch cart, plates, the cheerful coercion of nurses urging other patients to eat.}

\action{Kafka looks from the untouched tray to the proofs. One breath slips out through his nose. The corner of his mouth almost goes with it.}

\action{He picks up a slip of paper. Writes.}

\action{INSERT: ``The proofs are correct. Please proceed with publication.''}

\action{He pauses. Pencil still on the paper. Then adds a second line.}

\action{INSERT: ``Send a copy to Hermann Kafka, Prague.''}

\action{Dora, sitting beside the bed, sees the name.}

\character{DORA}
\paren{quietly}
\dialogue{Do you want him to read it?}

\action{Kafka looks at the slip. At his father's name in his own handwriting. The proofs on his lap. The untouched broth. The corridor outside where the lunch cart rattles.}

\action{He picks up the pencil. Draws one hard line through Hermann's name.}

\action{Dora says nothing. She saw the hesitation. The hesitation was longer than the line.}

\action{He folds the slip once. Unfolds it again. Leaves the crossed-out name visible, as if the act of hiding it would matter more than writing it.}

\action{His hand rests on the proof pages --- the last pages of his that will pass through his hands.}

\cutto

% ============================================================
% DORA — THE LAST TENDERNESS
% ============================================================

\scene{INT. SANATORIUM KIERLING -- KAFKA'S ROOM -- EVENING}

\action{The light is fading. Dora sits beside the bed. She is reading to him --- from a book of fairy tales, in German. Her voice is steady and warm.}

\action{Kafka listens. His eyes are closed but he is not asleep. His hand rests on the blanket. Dora's hand covers his.}

\action{She stops reading. Looks at him.}

\character{DORA}
\paren{very softly}
\dialogue{Franz?}

\action{He opens his eyes.}

\action{He picks up his pencil. Writes.}

\action{INSERT: ``Keep reading. Your voice makes the room bearable.''}

\action{Dora reads. He listens. The room darkens. Outside, birds. Inside, her voice.}

\cutto

% ============================================================
% FINAL VOICE / FINAL REFUSAL — OSCAR CLIP SCENE
% ============================================================

\scene{INT. SANATORIUM KIERLING -- KAFKA'S ROOM -- DAY}

\action{Morning. Late May. The room is bright with spring light. Kafka is worse. His throat is so swollen he cannot swallow water. Melted ice stands in a glass beside the bed. Morphine has been given, but not enough.}

\action{DR. HOFFMANN stands at the foot of the bed with his case open on a chair, cuffs unfastened. KLOPSTOCK stands beside a tray with the untouched broth, spoon, and glass. DORA has edged her chair so close her knees touch the bed. She tries the spoon once at Kafka's lips. He cannot take it.}

\action{Kafka writes. Shows the slip to Hoffmann.}

\action{INSERT: ``Give me morphine, Doctor. Otherwise you are a torturer.''}

\character{DR. HOFFMANN}
\dialogue{Herr Kafka, the dosage is already high. If I push it further, I may quiet more than the pain.}

\action{Kafka writes again, faster now. Shows it.}

\action{INSERT: ``I have been patient my entire life. Do not ask me to be patient with this.''}

\action{Hoffmann reads it. Takes off his spectacles. Cannot answer at once.}

\action{Kafka turns to Klopstock. Writes.}

\action{INSERT: ``Kill me, or else you are a murderer.''}

\action{Klopstock reads the note. Folds it once. Unfolds it. His hands shake.}

\character{KLOPSTOCK}
\paren{barely audible}
\dialogue{Franz...}

\action{Kafka looks at him and waits.}

\action{He reaches for the pencil again. Writes one more note, slower this time. Shows it to all three of them.}

\action{INSERT: ``No fear now.''}

\action{Dora takes his hand. He squeezes once. Klopstock turns away only far enough to steady himself.}

\action{Hoffmann prepares a larger dose with the controlled reluctance of a man who knows exactly what he is agreeing to. Kafka watches the needle go in.}

\action{The morphine reaches him. His face eases.}

\action{For a moment --- morphine-soft, impossible to verify --- the walls of the room are papered with manuscript pages. Claim forms. Notebook entries. Letters never sent. The nurse moves through the door and the pages hang like curtains around her.}

\action{Then the room is the room. The window. The garden. Spring light.}

\action{He opens his eyes once more. Looks at Dora. On her chair, half hidden beneath her shawl, the slip from the night before: ``It was the life with you. That was the real one.''}

\action{His hand moves to the pencil one last time. He writes on a slip, folds it once, and slides it under his pillow. Dora does not see what it says.}

\action{He closes his eyes.}

\cutto

% ============================================================
% THE DEATH
% ============================================================

\scene{INT. SANATORIUM KIERLING -- KAFKA'S ROOM -- AFTERNOON}

\titlecard{JUNE 3, 1924}

\action{The room. The bed. The window. Light.}

\action{Kafka lies still. Klopstock and Dora are present. The doctor has been called but has not arrived.}

\action{Klopstock holds Kafka's wrist. Counting. The count slows.}

\action{Dora holds Kafka's other hand. She is silent.}

\action{The room is very quiet. From outside: birdsong. The sound of wind in the garden. The sound of a world that continues.}

\action{Klopstock puts Kafka's hand down. Gently. He looks at Dora.}

\action{She knows.}

\action{Kafka's eyes are closed. Dora still holds his hand. Klopstock keeps looking at the wrist he no longer has to count.}

\action{Dora does not move. She sits holding his hand. The light in the room does not change.}

\fadetoblack

% ============================================================
% CODA: THE MANUSCRIPTS
% ============================================================

\titlecard{PRAGUE, 1924}

\scene{INT. KAFKA FAMILY APARTMENT -- LIVING ROOM -- DAY}

\action{Julie waits by the table with a box from Franz's room. Brod stands before her, hat in hand. On the table beside the box: the thick sealed packet addressed ``For Father.''}

\character{JULIE}
\dialogue{The notebooks were in the desk. This one was in mine.}

\action{Brod looks at the packet.}

\character{BROD}
\dialogue{He sent it back to Berlin.}

\character{JULIE}
\dialogue{I sent it. He put it in my hands years ago. Then it came back and sat in my drawer again.}

\action{A beat.}

\character{BROD}
\dialogue{Did Hermann ever see it?}

\character{JULIE}
\dialogue{No. Once he reached for the household ledger and almost touched the wrong drawer. I lied before he had the handle.}

\character{BROD}
\dialogue{Will you let me take it?}

\character{JULIE}
\dialogue{Take it, yes. Open it, no.}

\character{BROD}
\dialogue{Because it was for his father?}

\character{JULIE}
\dialogue{Because it was for an afternoon that never happened.}

\character{BROD}
\dialogue{He trusted you with that choice.}

\character{JULIE}
\dialogue{No. He put it in my hands and waited for me to choose which of them I could bear at supper.}

\action{Brod has no answer for that.}

\character{JULIE}
\dialogue{The rest he left among friends. This one he left with me.}

\action{She pushes the box toward him. Leaves the packet where it is a beat longer, then places it on top.}

\character{JULIE}
\dialogue{Do with the others what your conscience can survive. This one stays tied. If Franz had wanted it opened, he knew your address.}

\action{Brod nods. Takes the box and the packet.}

\scene{INT. MAX BROD'S APARTMENT -- NIGHT}

\action{Brod sits at his desk. Before him: stacks of manuscripts. Notebooks. Loose pages. The contents of Kafka's desk, delivered from Prague. Mixed among them: a Hebrew exercise page in Dora's hand and the folded Berlin registration paper, its purple stamp still visible at the edge. Separate from them: the thick sealed packet in Kafka's hand, addressed ``For Father.'' Beneath it, in smaller script: ``Via Mother.''}

\action{He touches the packet once. Reads the front again. He does not untie it.}

\action{He opens a notebook. Reads the first page. Then another page. Then another. His hand trembles slightly.}

\action{Among the notebooks, loose in the box: a single slip of paper, folded once. Smaller than the others. Dora has placed it on top with a note: ``Found under his pillow.''}

\action{Brod unfolds it. Reads it. His face goes still.}

\action{He refolds the slip and puts it in his breast pocket. He does not place it with the manuscripts. He does not place it with the letter.}

\action{At the edge of the desk: a matchbox. He picks it up. Weighs it in his hand.}

\action{He sets it beside a manuscript whose cover page reads, in Kafka's handwriting: ``DER PROZESS'' --- The Trial.}

\action{He opens the manuscript. Reads the first line. Stops. Reads it again.}

\action{Then he takes the matchbox, puts it in the desk drawer, and closes the drawer.}

\action{He pulls the next manuscript toward him.}

\action{The camera pulls back. Brod at his desk, surrounded by his dead friend's work. The lamp burning. The city outside.}

\cutto

% ============================================================
% FINAL IMAGE
% ============================================================

\scene{INT. APARTMENT -- BERLIN-STEGLITZ -- NIGHT}

\action{The room is empty now. The small table. Two chairs. The stove, cold.}

\action{Moonlight through the window falls on a Hebrew primer, one chipped bowl, the folded Berlin registration paper with the purple stamp still showing, and a small brass key.}

\action{The table drawer is shut. For the first time it closes all the way.}

\action{Through the window: Berlin. Bare trees. Tram wires. The narrow street below.}

\action{The table. The chairs. The light.}

\action{Hold.}
