% act3.tex — Movement III: The Women

\movement{III. THE WOMEN}

% ============================================================
% MEETING FELICE — AT BROD'S HOME
% ============================================================

\titlecard{PRAGUE, AUGUST 1912}

\scene{INT. MAX BROD'S APARTMENT -- EVENING}

\action{A warm, crowded apartment. Brod's family and friends talking over one another, reaching for bread, refilling glasses.}

\action{Kafka sits at the edge of it with a glass he has barely touched.}

\action{The bell. BROD crosses to the door. FELICE BAUER, 25, steps in with a mutual friend, removing her gloves as she enters. Practical dress. Direct eyes. She glances once at the mantel clock, more out of habit than worry.}

\action{Brod makes introductions.}

\character{BROD}
\dialogue{Felice, Franz Kafka. Insurance Institute by day. Trouble by night.}

\action{Felice extends her hand. Kafka takes it. Her grip is firm; his is brief.}

\character{FELICE}
\dialogue{What do you write?}

\character{KAFKA}
\dialogue{Stories. Short ones. About things going wrong.}

\character{FELICE}
\dialogue{For whom?}

\character{KAFKA}
\dialogue{Usually the man who thought he understood the room.}

\character{FELICE}
\dialogue{I work for a dictation machine company in Berlin. The Parlograph.}

\character{KAFKA}
\dialogue{A machine that records the human voice.}

\character{FELICE}
\dialogue{And plays it back. For business correspondence.}

\character{KAFKA}
\dialogue{A machine that remembers what one says. That's indecent.}

\action{Felice takes back her hand and keeps looking at him.}

\character{FELICE}
\dialogue{In Berlin we call it useful.}

\action{Kafka looks at her a second too long.}

\cutto

% ============================================================
% THE LETTERS BEGIN
% ============================================================

\scene{INT. KAFKA'S BEDROOM -- NIGHT}

\action{Kafka at his desk. A blank sheet where stories usually begin. Tonight it is a letter.}

\vo{KAFKA}{Dear Fräulein Bauer --- Do you even remember me? I am the thin man who sat across from you at Max Brod's table and who, piece by piece, passed you photographs of a trip to Thalia, which you were kind enough to look at.}

\action{He writes three lines, tears the page in half, starts again.}

\scene{INT. FELICE'S APARTMENT -- BERLIN -- DAY}

\action{A tidy apartment. Felice stands at a small desk in her hat and jacket, reading Kafka's letter before leaving for work.}

\action{She reads the last line again. One glove comes off. Then the hat. She sits back down and writes two quick pages before putting it on again.}

\scene{INT. PARLOGRAPH OFFICE -- BERLIN -- DAY}

\action{A modern office. Desks. Typewriters. Filing trays. A dictation machine on a demonstration table: polished metal, cylinder, mouthpiece.}

\action{Felice moves through it with brisk authority. Two YOUNG CLERKS trail her with order forms. A SALESMAN struggles with the machine's mouthpiece.}

\character{SALESMAN}
\dialogue{Fräulein Bauer, the Klein demonstration is at three and the cylinder keeps catching.}

\character{FELICE}
\dialogue{The cylinder is not catching. Herr Klein is turning it incorrectly. The machine is not temperamental. He is.}

\action{She takes the machine, adjusts one small screw, sets the cylinder, and speaks into the mouthpiece with efficient clarity.}

\character{FELICE}
\dialogue{Letter to Hirsch \& Co. Shipment confirmed for Thursday. Please stop telegraphing as if stationery were perishable.}

\action{The cylinder turns smoothly. The salesman stares. The young clerks hide smiles.}

\action{A MESSENGER brings in the afternoon post. Felice sorts it standing up. Bills. Orders. Circulars. One envelope from Prague in Kafka's careful hand. She pauses almost imperceptibly. Puts it aside. Finishes with the orders first.}

\scene{INT. KAFKA'S BEDROOM -- NIGHT}

\action{Another night. Another letter. The stack beside him has grown thick enough to need a paperweight.}

\vo{KAFKA}{I want to know what you eat for breakfast. Whether you sleep well. What the view from your office window looks like at four in the afternoon. If these are absurd questions, forgive me. They do not feel small here.}

\action{He seals the letter, adds it to the stack, then pulls a fresh page toward him before the wax has cooled.}

\scene{INT. FELICE'S APARTMENT -- BERLIN -- EVENING}

\action{Felice reads his latest letter at her kitchen table while eating dinner. Halfway through, she puts down her fork. Reads the same paragraph again.}

\action{The potatoes go cold. She folds the letter once, unfolds it, and keeps reading.}

\cutto

% ============================================================
% THE ENGAGEMENT — ESCALATION
% ============================================================

\scene{INT. ASKANISCHER HOF HOTEL -- BERLIN -- DAY}

\titlecard{BERLIN, 1914}

\action{A hotel meeting room. Wallpaper, water glasses, a long table too formal for an engagement.}

\action{Kafka sits on one side with Brod. Opposite: Felice between FRAU BAUER and ERNA. GRETE BLOCH at the far end, watchful.}

\action{Kafka wears his best suit. His hands are flat on the table.}

\character{FRAU BAUER}
\dialogue{Herr Kafka. We appreciate your interest in our Felice.}

\character{KAFKA}
\dialogue{Thank you, Frau Bauer.}

\character{FRAU BAUER}
\dialogue{You have a position at the Insurance Institute in Prague, I understand.}

\character{KAFKA}
\dialogue{Yes. A permanent position.}

\character{FRAU BAUER}
\dialogue{And what are your plans for advancement?}

\action{Kafka glances at Felice. She gives him the smallest nod: answer well.}

\character{KAFKA}
\dialogue{The Institute is stable. The salary is adequate.}

\character{FRAU BAUER}
\dialogue{That was not my question.}

\character{KAFKA}
\dialogue{I have not pursued advancement aggressively.}

\action{A pause. Under the table, Felice's hand moves toward his. She stops just short.}

\character{FRAU BAUER}
\dialogue{And where would my daughter live? In Prague?}

\character{KAFKA}
\dialogue{That was the idea.}

\character{FRAU BAUER}
\dialogue{Alone with you?}

\character{KAFKA}
\dialogue{No. At first, probably with my family until---}

\action{The room changes temperature. Felice closes her eyes once. Brod stares at the table.}

\character{FRAU BAUER}
\dialogue{And the writing?}

\character{KAFKA}
\dialogue{At night.}

\character{FRAU BAUER}
\dialogue{So my daughter lives in your parents' apartment while you write through the night.}

\character{KAFKA}
\dialogue{Temporarily.}

\character{FRAU BAUER}
\dialogue{Felice works hard. She expects order.}

\character{FELICE}
\paren{quietly firm}
\dialogue{Mother.}

\action{Frau Bauer stops. The room breathes.}

\action{Brod watches his friend. Grete watches Felice.}

\character{KAFKA}
\dialogue{Frau Bauer. Your concerns are not unfair.}

\action{That lands badly. He starts again.}

\character{KAFKA}
\dialogue{I can promise your daughter seriousness. I cannot promise comfort.}

\action{Silence. Frau Bauer looks at Felice. Felice keeps her face still. Kafka looks back down at his hands.}

\action{Grete lowers her eyes. She has heard enough.}

\cutto

% ============================================================
% THE ENGAGEMENT DINNER — OSCAR CLIP SCENE
% ============================================================

\scene{INT. RESTAURANT -- BERLIN -- EVENING}

\action{A private dining room. Bigger now: both families, friends, Brod, too many glasses, too much approval.}

\action{Kafka sits near the center of the table, smiling already before anyone speaks.}

\action{Across from him, Hermann is in his element --- booming, charming, taking up space. Julie watches Franz instead of the room.}

\character{HERMANN}
\paren{raising his glass}
\dialogue{My son! The doctor of law, the insurance man, and now a fiancé! Julie, can you believe it?}

\action{Laughter. Glasses lifted. Chairs shifting. Plates arriving.}

\action{Kafka raises his glass. Drinks. His hand is steady. His eyes are not.}

\action{Under the table, Felice takes his hand. He lets her. Her hand is warm and certain. His is cold.}

\action{CLOSE ON KAFKA'S FACE as the conversation swirls around him. Knife on plate. Laughter. Wine being poured. Hermann's laugh.}

\action{The sounds begin to align. Fork on china. A cough. Wine into glass. Another fork. Then a stamp inside his head.}

\action{Kafka's smile holds. Too correct.}

\character{FELICE}
\paren{whispering}
\dialogue{Are you all right?}

\character{KAFKA}
\paren{the smile never wavering}
\dialogue{Couldn't be better.}

\action{Felice looks at him, then back to her plate. She lets the answer stand.}

\character{HERMANN}
\paren{to Frau Bauer}
\dialogue{He'll settle now. You'll see. He needs a wife. Structure. Purpose. A woman like your Felice to keep him in line.}

\character{FRAU BAUER}
\dialogue{Felice is very organized. She runs the entire Berlin office.}

\character{HERMANN}
\paren{delighted}
\dialogue{There, you see? My boy has finally found some appetite!}

\action{He takes a dinner roll, tears it in half, eats, keeps talking.}

\action{Kafka opens his mouth. Something rises in him --- a sentence, an objection, the beginning of a door opening. Then it dies. His jaw closes. He sets down his fork.}

\action{Felice sees it. The thing that almost happened and didn't. She files it away without knowing what it was.}

\action{Later. Dessert. Coffee. The conversation has softened into the comfortable murmur of people who are satisfied with how things have gone.}

\action{Kafka excuses himself. Goes to the restroom.}

\scene{INT. RESTAURANT -- RESTROOM -- CONTINUOUS}

\action{He stands at the sink. Runs cold water. Holds his hands under it. Looks at himself in the mirror.}

\action{The smile stays on his face two seconds longer than it should. Then it drops away.}

\action{He grips the sink. Straightens his tie. Rebuilds the smile. Returns to the table.}

\action{The dinner continues. He sits. Felice squeezes his hand. He squeezes back.}

\cutto

% ============================================================
% THE BREAKING — FELICE
% ============================================================

\titlecard{SIX WEEKS LATER}

\scene{INT. HOTEL ROOM -- BERLIN -- DAY}

\action{A small hotel room. Neutral furniture. On the table between them: a neat stack of letters in Kafka's hand. Beside Felice's chair: a leather office satchel and a return train ticket.}

\action{Felice sits by the window in a dark dress, straight-backed. Kafka sits opposite, hands locked together.}

\character{FELICE}
\dialogue{Grete showed me your letters.}

\action{Kafka says nothing. Felice unties the ribbon around the stack.}

\character{FELICE}
\dialogue{In mine, you ask for marriage. In hers, you explain why it would destroy you.}

\character{FELICE}
\dialogue{You were very eloquent about it.\\
You're always eloquent about what you can't do.}

\character{KAFKA}
\paren{very quietly}
\dialogue{Felice---}

\character{FELICE}
\dialogue{No. Let me finish.}

\character{FELICE}
\dialogue{I kept thinking the letters were the difficult part. They weren't. They were the part you liked.}

\action{Kafka looks at his hands.}

\character{KAFKA}
\dialogue{I wanted it to work.}

\character{FELICE}
\dialogue{Work is what a man says when he wants credit for trying.}

\action{She slips the engagement ring from her finger and places it squarely on the top letter.}

\character{FELICE}
\dialogue{I came here from the office for this. I'll go back there after.}

\character{FELICE}
\dialogue{You wanted a woman in Berlin who stayed where you left her. Answering when addressed. Waiting when required.}

\character{FELICE}
\dialogue{The real me was opening the office at nine, answering telegrams, missing supper, sleeping badly. Your letters wanted me sitting still with time for them.}

\character{KAFKA}
\dialogue{That's not fair.}

\character{FELICE}
\dialogue{No. Fair would have been telling me sooner.}

\action{She stands. Puts on her coat. Buttons it carefully.}

\character{FELICE}
\dialogue{Goodbye, Franz.}

\action{She moves to the door. Stops.}

\character{FELICE}
\dialogue{You prefer wanting a thing to having it.}

\action{She leaves. The door closes. The room is silent.}

\action{Kafka does not move. The ring sits on the letters between them, where she left it.}

\cutto

% ============================================================
% THE KRAUS FILE RETURNS
% ============================================================

\scene{INT. INSURANCE INSTITUTE -- KAFKA'S OFFICE -- MORNING}

\action{Kafka at his desk in a fresh collar and the same dark suit. He has not slept enough to hide it.}

\action{He opens the top file. Inside: a follow-up report on the Kraus sawmill. A second worker injured. Different hand. Same machine. The replacement bolt has been removed again.}

\action{Kafka reads the report. Reads it again. His face hardens.}

\action{A knock. Marschner enters with a telegram already unfolded, coat still on as if three other doors have already kept him from going home.}

\character{MARSCHNER}
\dialogue{The owner says the worker ignored posted instructions. His counsel says we're confusing accident insurance with moral theater.}

\character{KAFKA}
\dialogue{A second man has lost the use of his hand to the same machine after a direct order.}

\character{MARSCHNER}
\dialogue{I know what the file says. By supper the Board will call it carelessness with witnesses.}

\action{Kafka stands. The gesture is abrupt enough to surprise even him.}

\character{KAFKA}
\dialogue{Then close the line.\\
Inspect every guard assembly in every Kraus mill.\\
Personal liability if one bolt is gone.}

\character{MARSCHNER}
\dialogue{Good. Now write it so Vienna can sign it.}

\action{Kafka sits again. Pulls a clean sheet toward him.}

\character{KAFKA}
\dialogue{Very well. `Repeated noncompliance creating foreseeable injury.'}

\character{MARSCHNER}
\dialogue{Yes. Cold enough to survive a meeting.}

\action{Marschner watches him begin. Some private satisfaction under the worry.}

\character{MARSCHNER}
\dialogue{By noon, Kafka. Before the owner has time to become sympathetic.}

\action{He leaves. Kafka writes. Ink catches at the nib. He keeps going.}

\action{The clock ticks. He does not look up.}

\cutto

% ============================================================
% THE AFTERMATH — ALONE
% ============================================================

\scene{INT. KAFKA'S BEDROOM -- NIGHT}

\action{Kafka at his desk. The lamp burns.}

\action{On the desk: Felice's letters, tied again with ribbon.}

\action{He opens his notebook. Writes.}

\action{He gets one line down. Stops. Tears out the page.}

\action{Through the wall: his father coughing in his sleep. Kafka waits for it to stop. It does not.}

\action{He turns the page. Starts again. This time it is not a letter and not a diary entry.}

\action{The words come fast.}

\action{He writes until the lamp burns low. Outside the window, Prague is dark. The spires of the Old Town are barely visible.}

\action{The pen keeps moving. The coughing goes on behind the wall.}

\cutto

% ============================================================
% MILENA — THE LETTERS
% ============================================================

\titlecard{PRAGUE / VIENNA, 1920}

\scene{INT. CAFÉ -- PRAGUE -- DAY}

\action{Post arrives at Kafka's café table. One envelope from Vienna, in an unfamiliar hand. He opens it.}

\vo{MILENA}{Dear Herr Kafka --- I have begun translating\\
your story ``The Stoker'' into Czech. I must tell you\\
your German is not what I expected. It is not\\
the German of a German. It is the German of someone\\
who has invented a language inside the language.\\
It is exact and strange. I cannot stop reading it.}

\action{Kafka reads it once, then again more slowly. His coffee goes cold beside his hand.}

\scene{INT. MILENA'S APARTMENT -- VIENNA -- DAY}

\action{MILENA JESENSKÁ, 24, works at a desk crowded with Kafka's pages, a Czech dictionary, a newspaper proof, and a cup gone cold. She pencils a sentence into the margin, checks the dictionary, recopies it cleanly, then tears the top half from another page and starts again.}

\action{ERNST POLAK passes through the room with his coat already on, an opened envelope in hand. He glances at the newspaper proof, then at Kafka's pages beneath it.}

\character{POLAK}
\dialogue{Tribuna wants the column before four. Your Prague author can wait.}

\character{MILENA}
\dialogue{The column will go at four. He won't.}

\action{Polak moves on. She keeps working.}

\scene{INT. KAFKA'S BEDROOM -- NIGHT}

\action{Kafka writes to Milena. No crossing out. One page, then another, before the first has dried.}

\vo{KAFKA}{I am afraid of everything, Milena. Of the dark, of the light, of what I might write, of what I might not write, of you, of not-you, of the distance between Prague and Vienna, which is too small to protect me and too large to cross.}

\cutto

% ============================================================
% MEETING MILENA — VIENNA
% ============================================================

\scene{EXT. VIENNA -- CAFÉ TERRACE -- AFTERNOON}

\action{Summer. An outdoor café. Linden trees. The sounds of Vienna: horses, trams, the particular music of a city that believes in pleasure.}

\action{Kafka waits at a table, aligning the spoon with the saucer, then misaligning it again.}

\action{Milena arrives with a pencil-marked translation page folded in her hand, a reporter's notebook under her arm, and ink on two fingers. She sets the page beside his cup, checks the café clock, sits, and says nothing until he looks up.}

\character{MILENA}
\dialogue{You look exactly like someone who should have stayed in Prague. Sit down. I have forty minutes before I hand in copy, so if you are going to be fascinating, do it quickly.}

\action{She sets the translation page between them, flags a sentence with her nail.}

\character{KAFKA}
\dialogue{Then I have traveled correctly.}

\character{MILENA}
\dialogue{I brought your page. You use five words where anyone reasonable would use eight and somehow it creates more trouble, not less. I have been fighting this sentence since Tuesday.}

\character{KAFKA}
\dialogue{Trouble is one of the cheaper effects.}

\character{MILENA}
\dialogue{Your stories make trouble. Your letters make it worse --- by the second page they have already started leaving the room. I keep translating them anyway because the German is so clean I can hear my own language underneath it, which is infuriating and also probably why I wrote to you in the first place.}

\action{Kafka reaches for his coffee. The spoon hits the cup. Milena notices, does not rescue him.}

\character{KAFKA}
\dialogue{Milena. I should tell you something.}

\character{MILENA}
\dialogue{You are going to tell me not to expect anything human from this.\\
Don't. I am a journalist. I expect copy and deadlines\\
and that men will disappoint me on schedule.}

\character{KAFKA}
\dialogue{I'm afraid. And I keep writing to you.\\
Maybe it's the same thing.}

\character{MILENA}
\dialogue{Of course it is. You disappear very early and keep signing your name. I noticed on page one.}

\action{He looks at her. This time he does not look away first.}

\cutto

% ============================================================
% MILENA — THE IMPOSSIBILITY
% ============================================================

\scene{EXT. VIENNA -- PARK -- LATE AFTERNOON}

\action{They walk. Close, but not touching.}

\character{KAFKA}
\dialogue{You're married.}

\character{MILENA}
\dialogue{Poorly.}

\character{KAFKA}
\dialogue{And I am--- }

\character{MILENA}
\dialogue{You are what? Finish the sentence. I finish mine.}

\character{KAFKA}
\dialogue{I am significantly worse in person.}

\character{MILENA}
\dialogue{Yes. Letters let you arrange yourself. In person you are nervous and your collar is crooked and you have not once looked at the trees, which are beautiful. Why are you here?}

\action{He stops walking. Looks at her.}

\character{KAFKA}
\dialogue{Because with you I thought I might fail in person and remain anyway.}

\action{She takes his hand. He lets her.}

\character{MILENA}
\dialogue{Then fail here. Stay. I have nowhere to be until four.}

\action{Traffic and voices continue around them. For once, Kafka does not retreat.}

\action{It lasts only a moment. A moment is more than he expected.}

\scene{EXT. VIENNA -- TRAIN PLATFORM -- EVENING}

\action{Steam. Porters. Calls from conductors. Kafka and Milena stand beside his carriage. He has not boarded yet.}

\action{Milena holds a folded timetable and one corrected proof page from his story. She presses both into his hand.}

\character{MILENA}
\dialogue{Ernst is in Dresden until Tuesday. There is a furnished room above my printer's shop on Štěpánská --- bad wallpaper, good lock, the tram stops outside. If you want an actual scandal, it is available and it has an address.}

\character{KAFKA}
\dialogue{You make catastrophe sound municipal.}

\character{MILENA}
\dialogue{I am a journalist. Everything is municipal. I am offering you a room, not a metaphor.}

\action{He looks at her. The noise of the station gives them privacy by refusing it.}

\character{KAFKA}
\dialogue{Prague is difficult.}

\character{MILENA}
\dialogue{Prague is your alibi. Prague has always been your alibi. I am standing on a platform in Vienna and I am asking about you.}

\character{KAFKA}
\dialogue{If you came closer, I would bring the whole apparatus with me. Fear. Delay. My father's voice. The rest of the furniture.}

\character{MILENA}
\dialogue{Franz, you already bring it. Only by post. At least in person I could see your face while you explain why you can't stay.}

\action{A whistle. Passengers begin boarding.}

\character{MILENA}
\dialogue{Do you know what I like least about your letters? They arrive having understood everything except whether you actually mean to live.}

\action{That lands. He cannot answer quickly enough.}

\character{MILENA}
\dialogue{Don't write tonight.\\
Let one thing happen without being turned into pages.}

\action{The conductor calls again. Kafka steps onto the carriage, then turns back.}

\character{KAFKA}
\dialogue{Milena---}

\character{MILENA}
\dialogue{I know. That's the trouble.}

\action{The train starts. Kafka remains at the door as the platform moves away. Milena does not wave.}
