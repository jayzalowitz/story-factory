% act1.tex — Movement I: The Father

\movement{I. THE FATHER}

% ============================================================
% OPENING: BERLIN, 1923 — THE BLOOD
% ============================================================

\titlecard{BERLIN, 1923}

\scene{EXT. BERLIN STREET -- STEGLITZ -- LATE AFTERNOON}

\action{A narrow street in winter. Bare trees. The light is already failing.}

\action{FRANZ KAFKA, 40, walks slowly along the pavement. He is thin in the way that suggests not asceticism but subtraction --- something has been taken from him steadily, over years. His overcoat hangs. His dark eyes are enormous in his angular face. Under one arm: a loaf wrapped in paper and a small packet of tea.}

\action{He passes a BUTCHER'S SHOP. In the window: hanging carcasses, pale and meticulously drained. A handwritten price card. Sawdust on the tile floor inside.}

\action{Kafka stops. Not because of the meat. Because something is wrong in his chest.}

\action{He coughs. Once. Twice. On the second cough he turns slightly from the window and covers his mouth with his hand.}

\action{When he takes his hand away, there is blood on his palm.}

\action{He stares at it. Behind him, through the glass, the butcher wraps a cut of beef in brown paper. The paper darkens where the blood seeps through.}

\action{Kafka looks at his hand. Looks at the window. The butcher's hands, working. His own hand, wet and red.}

\action{A WOMAN passes behind him on the pavement, shopping bag over her arm, and does not notice.}

\action{Kafka wipes his hand on his handkerchief. Folds it carefully. Puts it in his pocket.}

\action{He adjusts the paper parcel under his arm and keeps walking.}

\cutto

% ============================================================
% PRAGUE — THE FATHER'S WORLD
% ============================================================

\titlecard{PRAGUE, 1912}

\scene{INT. KAFKA FAMILY APARTMENT -- ALTSTÄDTER RING -- MORNING}

\action{A large, well-furnished apartment in the Old Town. Heavy furniture. Lace on the tables. Coffee already poured. Breakfast already arranged.}

\action{KAFKA, 29, sits at the table with bread in his hand and no intention of eating it.}

\action{JULIE KAFKA moves between stove and table, setting things right before they can go wrong. OTTLA eats across from Franz and nudges the jam toward him. He does not take it.}

\action{A door opens. The apartment shifts.}

\action{HERMANN KAFKA enters in a good suit, fastening a cuff. Large without softness.}

\character{HERMANN}
\dialogue{Is there more coffee, or did we drink the last of it yesterday and forget to order more?}

\action{Julie is already pouring. She sets the cup before him without a word. Hermann sits. The chair gives a small complaint under him. He unfolds his napkin.}

\character{HERMANN}
\paren{to Franz, without looking up}
\dialogue{You look terrible.}

\character{KAFKA}
\dialogue{I didn't sleep well.}

\character{HERMANN}
\dialogue{You never sleep well. You should eat more. You should exercise.}

\action{Hermann butters bread in four hard, practical strokes.}

\character{HERMANN}
\dialogue{What time did you get to bed?}

\character{KAFKA}
\dialogue{Late.}

\character{HERMANN}
\dialogue{Writing?}

\action{A pause. Kafka nods, barely.}

\character{HERMANN}
\dialogue{Your mother worries.}

\action{Julie, at the stove, keeps her back turned. Ottla stops eating.}

\character{HERMANN}
\dialogue{A man your age. A position at the Institute. You should be sleeping like a stone. Instead you sit up all night scribbling and then you come to breakfast looking like something the cat brought in.}

\action{Ottla looks at Franz. He does not look back.}

\character{KAFKA}
\paren{quietly}
\dialogue{I'll try to sleep earlier.}

\character{HERMANN}
\dialogue{Good. Now eat something.}

\action{Kafka takes a bite. Hermann watches until he swallows, then turns back to his coffee.}

\cutto

% ============================================================
% THE SHOP
% ============================================================

\scene{INT. HERMANN KAFKA'S FANCY GOODS SHOP -- DAY}

\action{A well-stocked retail shop on a good Prague street. Glass display cases. Buttons, ribbons, lace, parasols, walking sticks, gloves. Two SHOP GIRLS work behind the counter.}

\action{Hermann serves two customers at once, measuring ribbon for one while naming glove sizes for another without looking at the shelves.}

\action{Kafka enters. He is in his dark office suit, coming from the Insurance Institute. He carries a small package --- something Julie asked him to bring.}

\character{HERMANN}
\paren{to a departing customer}
\dialogue{Thank you, Mrs. Novotná. Always a pleasure.}

\action{He turns to Franz. His eyes go first to the collar, then the posture.}

\character{HERMANN}
\dialogue{Ah. Franz. Come, come. Put that in the back.}

\action{Kafka moves to the back room. One of the SHOP GIRLS, ANNA, looks up at him and then quickly back to her work.}

\action{Kafka returns from the back room. Hermann is already straightening a display of walking sticks by half-centimeters.}

\character{HERMANN}
\dialogue{Stand up straight.}

\action{Kafka straightens, slightly.}

\character{HERMANN}
\dialogue{Your mother wants to know about Saturday. Dinner. The Löwys are coming.}

\character{KAFKA}
\dialogue{I may have to work.}

\character{HERMANN}
\dialogue{On a Saturday.}

\character{KAFKA}
\dialogue{There's a case. The Kraus sawmill. A man lost three fingers in a---}

\character{HERMANN}
\paren{waving this away}
\dialogue{Fingers. Three fingers. Franz, the Löwys are family. You will come to dinner. You will wear a clean shirt. You will make conversation like a human being.}

\action{A CUSTOMER enters. Hermann turns at once, warm now, expansive, already smiling.}

\character{HERMANN}
\dialogue{Mr. Procházka! What a pleasure. Looking for something for your wife? We just received the most beautiful Italian leather gloves---}

\action{He moves away. Kafka stands alone among the glass cases and perfect rows of ribbon.}

\action{Anna glances at him again. He catches her eye.}

\character{KAFKA}
\paren{quietly, to Anna}
\dialogue{Do you ever feel the buttons are watching you?}

\action{Anna stares at him. The corner of her mouth gives way before she remembers where she is.}

\action{From across the shop, Hermann laughs. He turns the gloves in the light, finds the seam with one finger, and Mr. Procházka laughs too.}

\action{Kafka watches.}

\cutto

% ============================================================
% THE LAW DEGREE
% ============================================================

\scene{INT. KAFKA FAMILY APARTMENT -- LIVING ROOM -- EVENING}

\action{The apartment is set for a small celebration. Good plates. Bakery pastries. A bottle of wine already open.}

\action{Kafka sits with his Doctor of Law certificate in his lap. His sisters --- ELLI, VALLI, and OTTLA --- lean in to see it. Julie beams. Hermann stands with a full glass.}

\character{HERMANN}
\dialogue{Doctor of Law. My son.}

\action{He taps the certificate once with two fingers, proud.}

\character{HERMANN}
\dialogue{Now. We'll find you a practice. Or you'll stay at the Institute and advance. In five years, you'll be running a department.}

\character{KAFKA}
\dialogue{Perhaps.}

\character{HERMANN}
\dialogue{Not perhaps. Certainly. You have the degree. You have the mind. All you need is the---}

\action{He searches for the word. Makes a fist, gently.}

\character{HERMANN}
\dialogue{The appetite.}

\action{He takes a pastry from the plate and bites into it. Powdered sugar catches in his mustache. Kafka watches him eat.}

\action{Julie raises her glass.}

\character{JULIE}
\dialogue{To Franz.}

\action{They drink. Kafka raises his glass a beat later than the others.}

\action{Ottla touches his arm under the table. He looks at her. She squeezes once. He squeezes back.}

\cutto

% ============================================================
% THE NIGHT DESK
% ============================================================

\scene{INT. KAFKA'S BEDROOM -- NIGHT}

\action{Small. Austere. A narrow bed. A desk by the window. The family apartment is silent.}

\action{Kafka sits at the desk. It is after midnight. A single lamp burns. He is in his shirtsleeves, his jacket hung on the chair behind him.}

\action{He writes. Longhand, in ink, in the small careful script that is distinctly his.}

\action{We don't see what he writes. We see his face. The jaw unclenches. He leans closer to the page. For once he does not look toward the door.}

\action{From somewhere below, a sound: Hermann coughing in his sleep. The low rumble of a large man's restless body.}

\action{Kafka stops. Listens. His eyes go to the door. When nothing follows, he lowers the pen back to the page.}

\action{He writes again.}

\vo{KAFKA}{Dearest Father --- I have always been afraid of you, and yet I have never been able to explain why.}

\action{He stops. Crosses something out. Writes again.}

\action{The lamp flickers. Outside the window, Prague sleeps. The spires of the Old Town are barely visible in the dark.}

\action{Kafka writes. The pen moves. The only sound in the world.}

\cutto

% ============================================================
% THE MOTHER'S DIPLOMACY
% ============================================================

\scene{INT. KAFKA FAMILY APARTMENT -- KITCHEN -- MORNING}

\action{Early. Before Hermann wakes. Julie and Kafka at the kitchen table. Coffee. The intimacy of the early hour. Julie is slicing yesterday's bread thinner than bread should have to be sliced.}

\character{JULIE}
\dialogue{He doesn't mean it the way it sounds.}

\character{KAFKA}
\dialogue{I know.}

\character{JULIE}
\dialogue{He's proud of you. He just prefers pride when it can shout.}

\character{KAFKA}
\dialogue{I know, Mother.}

\action{A pause. Julie pushes the thinner slice toward him, keeps the heel for herself.}

\character{JULIE}
\dialogue{You're thin, Franz.}

\character{KAFKA}
\dialogue{I've always been thin.}

\character{JULIE}
\dialogue{Thinner.}

\action{She reaches across the table, closes the top button of his collar where it has worked loose, smooths it once with her thumb, and leaves her hand there until Hermann turns over in the next room. Only then does she take it back.}

\character{JULIE}
\dialogue{If you could just give him one sentence he can carry around the shop\\
without feeling laughed at, it would save trouble.}

\character{KAFKA}
\dialogue{Halfway to what?}

\action{Julie looks toward the bedroom door, then at the bread knife in her hand.}

\character{JULIE}
\dialogue{Halfway to breakfast.\\
Halfway through supper.\\
Halfway through one day without a campaign.}

\action{From the bedroom, the sound of Hermann waking: a great exhalation, the creak of the bed.}

\action{The day begins.}

\cutto

% ============================================================
% OTTLA
% ============================================================

\scene{EXT. PRAGUE -- VLTAVA RIVER EMBANKMENT -- AFTERNOON}

\action{Kafka and Ottla walk along the river. Autumn light on the water. The city behind them, golden and indifferent.}

\character{OTTLA}
\dialogue{I told Father I want to work on the farm.}

\character{KAFKA}
\dialogue{What did he say?}

\character{OTTLA}
\dialogue{What do you think he said?}

\action{They share a look. A small, grim amusement.}

\character{KAFKA}
\dialogue{He said you were being impractical.}

\character{OTTLA}
\dialogue{He said I was being ridiculous. That farming is for peasants and that he didn't build a business so his daughter could shovel manure.}

\character{KAFKA}
\dialogue{He has a way with imagery.}

\action{Ottla laughs. A real laugh, unguarded. Kafka drops his eyes, then the smile gets him anyway. His shoulders loosen before he can stop them.}

\character{OTTLA}
\dialogue{You should leave Prague, Franz.}

\character{KAFKA}
\dialogue{And go where?}

\character{OTTLA}
\dialogue{Anywhere. Berlin. Vienna. Somewhere he isn't standing in the next room.}

\action{Kafka looks at the river. The current moving steadily under the stone bridges.}

\character{KAFKA}
\dialogue{I've considered it.}

\character{OTTLA}
\dialogue{But?}

\character{KAFKA}
\dialogue{But I'd take him with me. He's not in the next room, Ottla. He's in this one.}

\action{He taps his chest. Lightly, almost playfully. But his eyes are serious.}

\character{OTTLA}
\dialogue{Then tell him. Not the one in here. The one at supper.}

\action{Kafka says nothing. He looks at the river instead.}

\action{They walk. The river moves beneath them, dark and purposeful, going somewhere they are not.}

\cutto

% ============================================================
% HERMANN'S WORLD — DINNER WITH THE LÖWYS
% ============================================================

\scene{INT. KAFKA FAMILY APARTMENT -- DINING ROOM -- EVENING}

\action{The dining room table is fully set. White cloth. Candles. Julie's best china. The LÖWY FAMILY --- an uncle, aunt, and cousin --- are seated. Hermann presides at the head.}

\action{The atmosphere is warm, prosperous, suffocating.}

\action{Hermann holds court. He tells a story about a business trip --- animated, commanding. The Löwys laugh in the right places.}

\action{Kafka sits at the corner of the table, eating carefully. He has arranged his food into small, precise portions. He takes tiny bites. His posture suggests a man trying to take up as little space as possible.}

\character{UNCLE LÖWY}
\dialogue{So, Franz. Your father tells us you're doing very well at the Institute.}

\character{KAFKA}
\dialogue{The Institute is--- efficient.}

\character{UNCLE LÖWY}
\dialogue{And the writing? Still at it?}

\action{A brief, heavy silence. Hermann's face does something complicated --- not a frown, but a recalibration.}

\character{HERMANN}
\dialogue{Franz writes in his spare time. As a hobby.}

\character{KAFKA}
\paren{very quietly}
\dialogue{Yes. A hobby.}

\action{Under the table, his hand tightens on his napkin.}

\character{UNCLE LÖWY}
\dialogue{What sort of stories?}

\character{KAFKA}
\dialogue{One man is arrested and never told why. Another wakes up an insect. The usual.}

\action{A silence. Uncle Löwy laughs, uncertainly. Hermann reaches for the bread basket.}

\character{AUNT LÖWY}
\dialogue{It's nice to have interests. My Heinrich paints watercolors. Very restful.}

\action{Kafka nods. Gives Aunt Löwy a brief smile, then drops his eyes to the table and reaches for his glass.}

\action{Hermann is already reaching with the bottle before anyone asks. Aunt Löwy lifts her glass to meet it. Uncle Löwy follows him into the toast. Julie slides the bread basket toward Hermann without looking up. Kafka watches how easily the whole table takes its shape from him.}

\character{HERMANN}
\dialogue{To family.}

\action{They raise their glasses. Kafka raises his. Drinks.}

\action{Later. The Löwys are gone. Julie clears the table. Hermann sits back, content.}

\character{HERMANN}
\dialogue{That went well. You see? No disaster.}

\action{Kafka stands in the doorway. Hermann sits broad and satisfied among the empty glasses. Kafka looks at him a long moment, thumb worrying the edge of the doorframe, then lets go.}

\character{KAFKA}
\dialogue{No. No disaster.}

\action{He goes to his room. Closes the door. Sits at his desk.}

\action{He does not write. He sits. The silence of the apartment settles around him like sediment.}

\action{Through the wall, he can hear his father humming as he prepares for bed. A tuneless, contented sound.}

\action{Kafka sits in the dark and listens to his father hum.}

\cutto

% ============================================================
% KAFKA ALONE — THE CORRIDOR
% ============================================================

\scene{INT. KAFKA FAMILY APARTMENT -- HALLWAY -- LATE NIGHT}

\action{The apartment is dark. Everyone is asleep.}

\action{Kafka moves through the hallway toward the bathroom. The hallway is narrow --- it has always been narrow. Tonight it keeps going. The walls press closer. His bare feet on the wooden floor answer him back.}

\action{He passes his parents' bedroom door. Behind it, Hermann's deep, regular breathing. A bedspring answers on the exhale.}

\action{The hallway stretches. Kafka walks. The bathroom door does not come nearer.}

\action{Then he is there. The hallway is normal. The door is where it always was.}

\action{He opens the door. Looks at his face in the mirror above the sink. Thin face. Dark eyes. Too awake.}

\action{He turns on the tap. Washes his hands. The water is cold.}

\action{In the mirror, the hallway behind him keeps going.}

\action{He turns off the tap. Goes back to bed.}
